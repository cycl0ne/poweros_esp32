# The Compute Module 4's config.txt, read by the VideoCore's firmware
# (start4.elf, beside it with fixup4.dat) from the eMMC's FAT partition -
# or from the PC over USB with the CM4-NANO-B's BOOT switch on - beside
# kernel.bin, disk-a64v80.bin and the module's device tree
# (bcm2711-rpi-cm4.dtb, Raspberry Pi's).

# The kernel at 2 MiB, where it copies itself from to where it is linked,
# 0x40200000 (src/arch/a64psci/kernel.ld, start.S). The device tree and
# the disk image apart above it: the image would be cleared with the
# kernel's .bss if it followed the kernel. The GIC on, as the armstub
# leaves it for a kernel that takes its interrupts there.
arm_64bit=1
kernel=kernel.bin
kernel_address=0x00200000
device_tree_address=0x03000000
initramfs disk-a64v80.bin 0x04000000
enable_gic=1
os_check=0

# The UARTs on, the PL011's clock with them: the PL011 is the kernel's
# raw port. On a module with the wireless chip the firmware gives GPIO 14
# and 15 - the IO board's header, pins 8 and 10 - to the mini UART, and
# undoes `gpio=` lines for them here: the kernel gives them to the PL011
# itself (src/boards/rpi_cm4/pins.zig). The CM4-NANO-B brings out
# neither; the HDMI port shows what the kernel says there
# (src/boards/rpi_cm4/screen.zig).
enable_uart=1

# The USB 2.0 port the BCM2711's xHCI's, not the DWC2's: host only, and
# the controller xhci.device drives (system.zig's USB part).
otg_mode=1

# The picture the firmware lights and hands the kernel: the whole of the
# monitor's preferred mode, no border.
disable_overscan=1
